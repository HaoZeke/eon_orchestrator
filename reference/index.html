6.2 Optimization¶

neb:
  optimization:
    images: 18
    max_iterations: 1000
    converged_force: 0.0514221
    opt_method: lbfgs
    max_move: 0.1

    # Energy-weighted springs
    energy_weighted: true
    ew_ksp_min: 0.972  # eV/Å²
    ew_ksp_max: 9.72   # eV/Å²
    ew_trigger: 0.5    # eV

    # Climbing image
    climbing_image_method: true
    ci_after_rel: 0.8

    # MMF refinement
    ci_mmf: true
    ci_mmf_nsteps: 1000

    # IDPP initializer
    sidpp_growth_alpha: 0.33
